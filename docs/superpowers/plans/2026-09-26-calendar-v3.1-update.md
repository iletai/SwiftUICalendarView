```swift
    /// Computes 7 day states for the week containing `date`, respecting `calendar.firstWeekday`.
    public static func computeWeekMatrix(
        for date: Date,
        calendar: Calendar,
        todayDate: Date = Date(),
        selectedKeySet: Set<Int> = [],
        rangeBounds: (startKey: Int?, endKey: Int?) = (nil, nil),
        disabledRule: (@Sendable (Date) -> Bool)? = nil,
        eventsMap: [Int: [CalendarEvent]] = [:]
    ) -> [CalendarDayState] {
        let cal = calendar
        let weekday = cal.component(.weekday, from: date)
        let offset = (weekday - cal.firstWeekday + 7) % 7
        guard let weekStart = cal.date(byAdding: .day, value: -offset, to: date) else { return [] }

        let todayKey = CalendarDateKey.key(for: todayDate, using: cal)
        let monthOfDate = cal.component(.month, from: date)
        let yearOfDate = cal.component(.year, from: date)

        var days: [CalendarDayState] = []
        days.reserveCapacity(7)

        for i in 0..<7 {
            guard let current = cal.date(byAdding: .day, value: i, to: weekStart) else { continue }
            let comps = cal.dateComponents([.year, .month, .day], from: current)
            let y = comps.year ?? 0
            let m = comps.month ?? 0
            let d = comps.day ?? 0
            let key = CalendarDateKey.makeKey(year: y, month: m, day: d)

            let isCurrentMonth = (y == yearOfDate && m == monthOfDate)
            let isToday = (key == todayKey)
            let isSelected = selectedKeySet.contains(key)
            let isDisabled = disabledRule?(current) ?? false

            var pos: RangeSelectionPosition = .none
            if let start = rangeBounds.startKey, let end = rangeBounds.endKey {
                if start == end && key == start {
                    pos = .singleDay
                } else if key == start {
                    pos = .rangeStart
                } else if key == end {
                    pos = .rangeEnd
                } else if key > start && key < end {
                    pos = .rangeMiddle
                }
            } else if let start = rangeBounds.startKey, key == start {
                pos = .singleDay
            }

            days.append(CalendarDayState(
                dateKey: key,
                date: current,
                dayNumber: d,
                isCurrentMonth: isCurrentMonth,
                isToday: isToday,
                isSelected: isSelected,
                isSelectable: !isDisabled,
                isDisabled: isDisabled,
                rangePosition: pos,
                events: eventsMap[key] ?? []
            ))
        }
        return days
    }
```

```swift
            if state.rangePosition != .none {
                RangeSelectionShape(position: state.rangePosition, cornerRadius: 8)
                    .fill(
                        state.rangePosition == .rangeStart || state.rangePosition == .rangeEnd || state.rangePosition == .singleDay
                        ? AnyShapeStyle(LinearGradient(
                            colors: [accentColor, accentColor.opacity(0.8)],
                            startPoint: .topLeading,
                            endPoint: .bottomTrailing
                        ))
                        : AnyShapeStyle(accentColor.opacity(0.15))
                    )
            } else if state.isSelected {
                Circle()
                    .fill(LinearGradient(
                        colors: [accentColor, accentColor.opacity(0.8)],
                        startPoint: .topLeading,
                        endPoint: .bottomTrailing
                    ))
            } else if state.isToday {
                Circle()
                    .stroke(accentColor, lineWidth: 1.5)
                    .shadow(color: accentColor.opacity(0.3), radius: 4)
            }

            // Content
            VStack(spacing: 2) {
                Text("\(state.dayNumber)")
                    .font(.system(size: 16, weight: state.isSelected || state.isToday ? .semibold : .regular))
                    .foregroundColor(textColor)

                if !state.events.isEmpty {
                    HStack(spacing: 2) {
                        ForEach(state.events.prefix(3), id: \.id) { event in
                            Circle()
                                .fill(event.color)
                                .frame(width: 4, height: 4)
                        }
                    }
                }
            }
        }
        .frame(height: 38)
        .opacity(state.isCurrentMonth ? 1.0 : 0.3)
        .scaleEffect(state.isSelected ? 1.0 : 0.97)
        .animation(.spring(response: 0.25, dampingFraction: 0.7), value: state.isSelected)
        .accessibilityElement(children: .ignore)
        .accessibilityLabel(accessibilityText)
        .accessibilityAddTraits(state.isSelected ? [.isSelected, .isButton] : [.isButton])
    }

    public var textColor: Color {
        if state.isSelected {
            return .white
        }
        if state.isDisabled {
            return .secondary.opacity(0.5)
        }
        return state.isCurrentMonth ? .primary : .secondary
    }

    public var accessibilityText: String {
        let dateStr = state.date.formatted(.dateTime.year().month().day())
        let eventStr = state.events.isEmpty ? "" : ", \(state.events.count) events"
        return "\(dateStr)\(eventStr)"
    }
```
